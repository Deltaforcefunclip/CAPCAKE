cmake_minimum_required(VERSION 3.18)
project(rifejni)
set(CMAKE_CXX_STANDARD 17)
set(ncnn_DIR ${CMAKE_CURRENT_SOURCE_DIR}/ncnn/${ANDROID_ABI}/lib/cmake/ncnn)
find_package(ncnn REQUIRED)
file(GLOB SRC ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_library(rifejni SHARED ${SRC})
target_include_directories(rifejni PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(rifejni ncnn android log)
